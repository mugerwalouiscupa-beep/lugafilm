-- 2) Viewer account control table. create table if not exists public.lugafilm_viewer_accounts ( user_id uuid primary key references auth.users(id) on delete cascade, email text, status text not null default 'active', reason text, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); create index if not exists lugafilm_viewer_accounts_status_idx on public.lugafilm_viewer_accounts(status); -- 3) RLS for subscription requests. alter table public.lugafilm_subscriptions enable row level security; drop policy if exists "lugafilm subscriptions admin update" on public.lugafilm_subscriptions; create policy "lugafilm subscriptions admin update" on public.lugafilm_subscriptions for update to authenticated using ( exists ( select 1 from public.lugafilm_admins a where a.user_id = auth.uid() ) ) with check ( exists ( select 1 from public.lugafilm_admins a where a.user_id = auth.uid() ) ); drop policy if exists "lugafilm subscriptions admin delete" on public.lugafilm_subscriptions; create policy "lugafilm subscriptions admin delete" on public.lugafilm_subscriptions for delete to authenticated using ( exists ( select 1 from public.lugafilm_admins a where a.user_id = auth.uid() ) );
